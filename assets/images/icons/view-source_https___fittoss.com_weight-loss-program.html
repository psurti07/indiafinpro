const PRODUCT_ID = 1;
window.otpSubmitting = false;

$(document).ready(function() {
const $form = $('#basic-webinar-details-form');
const otpInputs = document.querySelectorAll('.otp');

function getOtpValue() {
return Array.from(otpInputs).map(i => i.value).join('');
}

function tryAutoVerify() {
const otp = getOtpValue();
const allFilled = Array.from(otpInputs).every(i => i.value !== '');
if (otp.length === otpInputs.length && allFilled) {
setTimeout(() => {
$('#btn-verify-otp').trigger('click');
}, 100);
}
}

otpInputs.forEach((input, index) => {

input.addEventListener('input', function() {
this.value = this.value.replace(/[^0-9]/g, '');

if (this.value && index < otpInputs.length - 1) {
otpInputs[index + 1].focus();
}

tryAutoVerify();
});

input.addEventListener('keydown', function(e) {
if (e.key === 'Backspace' && !this.value && index > 0) {
otpInputs[index - 1].focus();
}
});

input.addEventListener('paste', function(e) {
e.preventDefault();
const pasted = e.clipboardData.getData('text').replace(/\D/g, '');

if (pasted.length === otpInputs.length) {
pasted.split('').forEach((digit, i) => {
otpInputs[i].value = digit;
});
otpInputs[otpInputs.length - 1].focus();
tryAutoVerify();
}
});
});

$form.on('submit', function(e) {
e.preventDefault();
if ($('#otp_verified_flag').val() !== '1') {
e.preventDefault();
return false;
}

if (!window.validateOfferForm()) return;

const formData = new FormData(this);
const $btn = $('#go-step-2');
$btn.prop('disabled', true).html(
'<span class="spinner-border spinner-border-sm"></span> Please Wait...'
);

$.ajax({
url: $form.attr('action'),
type: 'POST',
data: formData,
processData: false,
contentType: false,
headers: {
'X-CSRF-TOKEN': $('meta[name="csrf-token"]').attr('content')
},
success: function(res) {
$btn.prop('disabled', false).html(
'Start Now <i class="icon-arrow-top-right"></i>'
);

if (res.type === 'SUCCESS' || res.type === 'INFO') {
toastr.success(res.message);

if (res.redirect) {
$btn.prop('disabled', true).html('Redirecting...');
setTimeout(() => {
$btn.prop('disabled', false).html('Start Now <i class="icon-arrow-top-right"></i>');
window.location.href = res.redirect
}, 1000);
}
} else {
toastr.error(res.message);
}
},
error: function(xhr) {
// $('#mobile_no').prop('readonly', false);
$btn.prop('disabled', false).html(
'Let\'s Start <i class="icon-arrow-top-right"></i>'
);

if (xhr.status === 422 && xhr.responseJSON?.errors) {
$.each(xhr.responseJSON.errors, function(field, messages) {

$('.' + field).html('<strong>' + messages[0] +
'</strong>');

toastr.error(messages[0]);
});
} else if (xhr.responseJSON?.message) {
toastr.error(xhr.responseJSON.message);
} else {
toastr.error('Something went wrong. Please try again.');
}
}
});
});

$("#mobile_no").on("input", function() {
this.value = this.value.replace(/[^0-9]/g, "");
});

$('#step-1').fadeIn();
});

if (!window.validateOfferForm()) return;

let mobile_no = $('#mobile_no').val();

const $btn = $(this);
$btn.prop('disabled', true).html('Please wait...');

$.post("https://fittoss.com/send-otp", {
mobile: mobile_no,
product_id: PRODUCT_ID,
_token: $('meta[name="csrf-token"]').attr('content')

$('#step-1').hide();

$('#step-2').fadeIn();
} else {
// $('#mobile_no').prop('readonly', false);

toastr.error(res.message);
}
}).fail(function(xhr) {
if (xhr.status === 422) {
let errors = xhr.responseJSON.errors;
if (errors.mobile) {
toastr.error(errors.mobile[0]);
} else {
toastr.error('Validation failed');
}
} else {
toastr.error('Something went wrong');
}
})
.always(function() {

.html('Start Now <i class="icon-arrow-top-right"></i>');
});

},
error: function(xhr) {

if (xhr.responseJSON?.message) {
toastr.error(xhr.responseJSON.message);

toastr.error('The OTP is invalid or has expired. Please try again.');
}
$btn.prop('disabled', false).html('Verify OTP <i class="icon-arrow-top-right"></i>');
}
});
});
